🚀 Quick Start

# Install
pip install -e .

# Initialize database
jeera-trader db --init

# Collect data (FREE!)
jeera-trader collect --source all --start-date 2024-01-01 --end-date 2024-12-31

# Generate features
jeera-trader features --start-date 2024-01-01 --end-date 2024-12-31

# Train model
jeera-trader train --model rf

# Get trading signal
jeera-trader signal
